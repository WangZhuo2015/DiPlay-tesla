set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_INSTALL_RULES ON)
# Share immutable dependency sources between Android ABIs and build types.
set(FETCHCONTENT_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/rtc-deps")
file(MAKE_DIRECTORY "${FETCHCONTENT_BASE_DIR}")
set(FETCHCONTENT_UPDATES_DISCONNECTED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)

set(USE_MBEDTLS ON CACHE BOOL "" FORCE)

# WebRTC requires the DTLS use_srtp extension.
add_compile_definitions(MBEDTLS_SSL_DTLS_SRTP)
include(NativeDependencies.cmake)
function(diplay_populate name)
    # AGP configures ABIs concurrently. Serialize source extraction only; binary trees stay separate.
    file(LOCK "${FETCHCONTENT_BASE_DIR}/populate.lock" GUARD FUNCTION TIMEOUT 300)
    set(archive "${DIPLAY_${name}_URL}")
    if(DEFINED ENV{DIPLAY_RTC_ARCHIVE_DIR} AND EXISTS "$ENV{DIPLAY_RTC_ARCHIVE_DIR}/${name}.tar.gz")
        set(archive "$ENV{DIPLAY_RTC_ARCHIVE_DIR}/${name}.tar.gz")
        file(TO_CMAKE_PATH "${archive}" archive)
    endif()
    set(destination "${FETCHCONTENT_BASE_DIR}/${name}-src")
    if(ARGC GREATER 1)
        set(destination "${ARGV1}")
    endif()
    set(binary "${CMAKE_BINARY_DIR}/deps/${name}")
    set(expected "${DIPLAY_${name}_SHA256}")
    set(actual "")
    if(EXISTS "${destination}/.diplay-populated")
        file(READ "${destination}/.diplay-populated" actual)
    endif()
    if(NOT actual STREQUAL expected)
        FetchContent_Declare(${name} URL "${archive}" URL_HASH "SHA256=${expected}" SOURCE_DIR "${destination}" BINARY_DIR "${binary}")
        FetchContent_Populate(${name})
        file(WRITE "${destination}/.diplay-populated" "${expected}")
    endif()
    set(${name}_SOURCE_DIR "${destination}" PARENT_SCOPE)
    set(${name}_BINARY_DIR "${binary}" PARENT_SCOPE)
endfunction()
diplay_populate(mbedtls)
diplay_populate(framework "${mbedtls_SOURCE_DIR}/framework")
add_subdirectory("${mbedtls_SOURCE_DIR}" "${mbedtls_BINARY_DIR}")
add_library(MbedTLS::MbedTLS ALIAS mbedtls)
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
diplay_populate(libdatachannel)
foreach(dependency json libjuice libsrtp plog usrsctp)
    diplay_populate(${dependency} "${libdatachannel_SOURCE_DIR}/deps/${dependency}")
endforeach()
add_subdirectory("${libdatachannel_SOURCE_DIR}" "${libdatachannel_BINARY_DIR}")
add_library(diplay-rtc SHARED rtc_bridge.cpp)
target_link_libraries(diplay-rtc PRIVATE datachannel-static log)
target_link_options(diplay-rtc PRIVATE "-Wl,-z,max-page-size=16384")
